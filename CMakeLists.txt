cmake_minimum_required(VERSION 3.24)
project(WardogsKillTracker VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(NOT WIN32)
  message(FATAL_ERROR "WardogsKillTracker supports Windows only")
endif()
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc imgcodecs highgui)
include(FetchContent)
find_package(nlohmann_json CONFIG QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
  FetchContent_MakeAvailable(json)
endif()
add_executable(WardogsKillTracker
  src/main.cpp src/Application.cpp src/CenterKillDetector.cpp src/Config.cpp src/Logger.cpp src/RecordingMonitor.cpp
  src/ScreenCapture.cpp src/KillDetector.cpp src/Session.cpp src/Utils.cpp)
target_include_directories(WardogsKillTracker PRIVATE src)
target_compile_definitions(WardogsKillTracker PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(WardogsKillTracker PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(WardogsKillTracker PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json d3d11 dxgi user32 shell32)
add_custom_command(TARGET WardogsKillTracker POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_SOURCE_DIR}/config.json $<TARGET_FILE_DIR:WardogsKillTracker>/config.json
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:WardogsKillTracker>/assets)
include(CTest)
if(BUILD_TESTING)
 add_executable(WardogsKillTrackerTests tests/tests.cpp src/Utils.cpp src/Config.cpp src/Session.cpp src/CenterKillDetector.cpp src/KillDetector.cpp)
 target_include_directories(WardogsKillTrackerTests PRIVATE src)
 target_link_libraries(WardogsKillTrackerTests PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
 target_compile_options(WardogsKillTrackerTests PRIVATE /UNDEBUG /EHsc)
 target_compile_definitions(WardogsKillTrackerTests PRIVATE TEST_ASSET_DIR="${CMAKE_SOURCE_DIR}/assets")
 target_compile_definitions(WardogsKillTrackerTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 add_test(NAME core COMMAND WardogsKillTrackerTests)
endif()
